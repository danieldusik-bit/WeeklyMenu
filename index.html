function toggleDay(dayIdx) {

    const card = document.getElementById(`day-${dayIdx}`);

    const index = openDays.indexOf(dayIdx);

    if (index > -1) {

        openDays.splice(index, 1);

        card.classList.remove('expanded');

    } else {

function renderFilters() {

    const container = document.getElementById('filter-container');

    container.innerHTML = categories.map(cat => `

        <button class="filter-chip ${shopFilter === cat ? 'active' : ''}" onclick="setFilter('${cat}')">${cat}</button>

    `).join('');

}


function setFilter(cat) {

    shopFilter = cat;

    renderFilters();

    renderShopping();

}


function renderShopping() {

    const container = document.getElementById('shopping-container');

    let html = '';


    const filteredCategories = shopFilter === "All" ? categories.slice(1) : [shopFilter];

    

    let totalItems = 0;

    let totalChecked = 0;


    filteredCategories.forEach(cat => {

        const items = shoppingItems.filter(item => item.cat === cat);

        if (items.length === 0) return;


        html += `

            <div class="shop-category">

                <div class="shop-category-title">${cat}</div>

        `;


        items.forEach(item => {

            const isChecked = checkedItems.includes(item.id);

            totalItems++;

            if (isChecked) totalChecked++;


            html += `

                <div class="shop-item ${isChecked ? 'checked' : ''}" onclick="toggleItem(${item.id})">

                    <div class="checkbox"></div>

                    <div class="shop-item-info">

                        <div class="shop-item-name">${item.name} <span class="shop-item-qty">· ${item.qty}</span></div>

                        <div class="shop-item-meta">${item.meals.join(', ')}</div>

                    </div>

                </div>

            `;

        });


        html += `</div>`;

    });


    container.innerHTML = html;


    // Update progress only if "All" is selected to show accurate total progress

    if (shopFilter === "All") {

        document.getElementById('progress-count').textContent = `${totalChecked} / ${totalItems}`;

        const pct = totalItems === 0 ? 0 : (totalChecked / totalItems) * 100;

        document.getElementById('progress-bar').style.width = `${pct}%`;

    }

}


function toggleItem(id) {

    const index = checkedItems.indexOf(id);

    if (index > -1) {

        checkedItems.splice(index, 1);

    } else {

        checkedItems.push(id);

    }

    renderShopping();

}


function resetShop() {

    if (confirm("Clear all checked items?")) {

        checkedItems = [];

        renderShopping();

    }

}
